import React, { useState } from 'react';

export default function Login() {
  
  // Nossos Estados (variáveis dinâmicas)
  const [usuario, setUsuario] = useState("");
  const [senha, setSenha] = useState("");
  const [erro, setErro] = useState("");

  // Função disparada ao submeter o formulário
  const handleLogin = (evento) => {
    evento.preventDefault(); 
    setErro(""); 

    if (usuario === "admin" && senha === "123") {
        alert("Login realizado com sucesso!");
    } else {
        setErro("Usuário ou senha inválidos!");
    }
  };

  // Função para o link de esquecer a senha
  const handleEsqueceuSenha = (evento) => {
    evento.preventDefault(); 
    alert("Um link de recuperação de senha foi enviado para o seu e-mail!");
  };

  return (
    <div className="login-container">
      <div className="card">
        <h2>Login</h2>
        
        <form onSubmit={handleLogin}>
          
          <input 
            type="text" 
            placeholder="Usuário" 
            value={usuario} 
            onChange={(evento) => setUsuario(evento.target.value)} 
            required 
          />
          
          <input 
            type="password" 
            placeholder="Senha" 
            value={senha} 
            onChange={(evento) => setSenha(evento.target.value)} 
            required 
          />
          
          {/* Trocamos o 'style' direto por uma 'className' chamada 'mensagem-erro' */}
          {erro && <div className="mensagem-erro">{erro}</div>}
          
          <button type="submit">Entrar</button>

          {/* Trocamos os estilos do container e do link por classes CSS */}
          <div className="esqueceu-senha-container">
            <a 
              href="#" 
              onClick={handleEsqueceuSenha} 
              className="esqueceu-senha-link"
            >
              Esqueceu a senha?
            </a>
          </div>

        </form>
      </div>
    </div>
  );
}

import React, { useState } from 'react';
// IMPORTAÇÃO NOVA: Trazemos o gancho de navegação do React Router
import { useNavigate } from 'react-router-dom';

export default function Login() {
  
  const [usuario, setUsuario] = useState("");
  const [senha, setSenha] = useState("");
  const [erro, setErro] = useState("");

  // CRIAMOS O NAVEGADOR: Essa variável agora tem o poder de mudar a página
  const navigate = useNavigate();

  const handleLogin = (evento) => {
    evento.preventDefault(); 
    setErro(""); 

    if (usuario === "admin" && senha === "123") {

        // Salvamos o "crachá" no navegador do usuário antes de viajar para a Dashboard
        localStorage.setItem("auth", "true");
        navigate("/dashboard");
    } else {
        setErro("Usuário ou senha inválidos!");
    }
  };

  const handleEsqueceuSenha = (evento) => {
    evento.preventDefault(); 
    alert("Um link de recuperação de senha foi enviado para o seu e-mail!");
  };

  return (
    <div className="login-container">
      <div className="card">
        <h2>Login</h2>
        
        <form onSubmit={handleLogin}>
          
          <input 
            type="text" 
            placeholder="Usuário" 
            value={usuario} 
            onChange={(evento) => setUsuario(evento.target.value)} 
            required 
          />
          
          <input 
            type="password" 
            placeholder="Senha" 
            value={senha} 
            onChange={(evento) => setSenha(evento.target.value)} 
            required 
          />
          
          {erro && <div className="mensagem-erro">{erro}</div>}
          
          <button type="submit">Entrar</button>

          <div className="esqueceu-senha-container">
            <a 
              href="#" 
              onClick={handleEsqueceuSenha} 
              className="esqueceu-senha-link"
            >
              Esqueceu a senha?
            </a>
          </div>

        </form>
      </div>
    </div>
  );
}
